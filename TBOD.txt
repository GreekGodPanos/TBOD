local function colorSeq(...)
	local keys = {}
	for _, k in ipairs({...}) do
		keys[#keys + 1] = ColorSequenceKeypoint.new(k[1], Color3.fromRGB(k[2], k[3], k[4]))
	end
	return ColorSequence.new(keys)
end

local function numSeq(...)
	local keys = {}
	for _, k in ipairs({...}) do
		keys[#keys + 1] = NumberSequenceKeypoint.new(k[1], k[2])
	end
	return NumberSequence.new(keys)
end

local function recolorSequence(sequence, color)
	local keys = {}
	for _, k in ipairs(sequence.Keypoints) do
		local c = k.Value
		local whiteness = math.min(c.R, c.G, c.B)
		local brightness = math.max(c.R, c.G, c.B)
		local tinted = color:Lerp(Color3.new(1, 1, 1), whiteness)
		keys[#keys + 1] = ColorSequenceKeypoint.new(k.Time, Color3.new(tinted.R * brightness, tinted.G * brightness, tinted.B * brightness))
	end
	return ColorSequence.new(keys)
end

local function parseColor(text)
	text = (text:gsub("%s", ""))
	if text == "" or text:lower() == "default" then return nil, true end
	local hex = text:match("^#?(%x%x%x%x%x%x)$")
	if hex then
		return Color3.fromRGB(tonumber(hex:sub(1, 2), 16), tonumber(hex:sub(3, 4), 16), tonumber(hex:sub(5, 6), 16)), true
	end
	local r, g, b = text:match("^(%d+),(%d+),(%d+)$")
	if r then
		return Color3.fromRGB(math.clamp(tonumber(r), 0, 255), math.clamp(tonumber(g), 0, 255), math.clamp(tonumber(b), 0, 255)), true
	end
	return nil, false
end

local function colorToText(color)
	if not color then return "default" end
	return string.format("#%02X%02X%02X", math.floor(color.R * 255 + 0.5), math.floor(color.G * 255 + 0.5), math.floor(color.B * 255 + 0.5))
end

local function resolvePath(pathParts)
	local current = game
	for _, name in ipairs(pathParts) do
		current = current:FindFirstChild(name)
		if not current then return nil end
	end
	return current
end

local function parseSuffixValue(str)
	if not str then return 0 end
	str = string.upper(string.gsub(str, "[%,%$%s]", ""))
	local num, suffix = string.match(str, "([%d%.]+)(%a*)")
	num = tonumber(num) or 0
	local mult = suffix and CONFIG.Suffixes[suffix]
	return mult and num * mult or num
end

local FORMAT_STEPS = { { 1e18, "Qi" }, { 1e15, "Qa" }, { 1e12, "T" }, { 1e9, "B" }, { 1e6, "M" }, { 1e3, "K" } }
local function formatNumber(num)
	for _, step in ipairs(FORMAT_STEPS) do
		if num >= step[1] then return string.format("%.2f%s", num / step[1], step[2]) end
	end
	return string.format("%.0f", num)
end

local function parsePrice(str)
	if not str or str == "" then return nil end
	local cleanStr = str:gsub("[$,]", ""):lower()
	if cleanStr:find("free") then return 0 end
	local num = tonumber(cleanStr:match("[%d%.]+"))
	if not num then return nil end
	if cleanStr:find("k") then num = num * 1e3
	elseif cleanStr:find("m") then num = num * 1e6
	elseif cleanStr:find("b") then num = num * 1e9
	elseif cleanStr:find("t") then num = num * 1e12
	elseif cleanStr:find("qa") then num = num * 1e15
	elseif cleanStr:find("qi") then num = num * 1e18 end
	return num
end

local function parseProgress(text)
	if string.find(string.upper(text), "COMPLETE") then return 100 end

	local currentStr, maxStr = string.match(text, "([%d%,%.%a]+)%s*/%s*([%d%,%.%a]+)")
	if currentStr and maxStr then
		local current = parseSuffixValue(currentStr)
		local max = parseSuffixValue(maxStr)
		local leaderstats = localPlayer:FindFirstChild("leaderstats")
		local cashVal = leaderstats and (leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Money"))
		if cashVal then
			local rawCash = tonumber(cashVal.Value) or 0
			if rawCash > current then current = rawCash end
		end
		if max > 0 then return math.clamp((current / max) * 100, 0, 100) end
	end

	local rawPercent = string.match(text, "(%d+%.?%d*)%%")
	if rawPercent then return math.clamp(tonumber(rawPercent) or 0, 0, 100) end
	return 0
end

local function monitorText(key, pathParts)
	task.spawn(function()
		local activeConnection
		while running do
			local obj = resolvePath(pathParts)
			if obj then
				local actualText
				if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
					actualText = obj.Text
				elseif obj:IsA("StringValue") then
					actualText = obj.Value
				else
					pcall(function() actualText = obj.Text end)
				end

				if actualText ~= nil then
					trackedTexts[key] = tostring(actualText)
					if not activeConnection then
						local prop = obj:IsA("StringValue") and "Value" or "Text"
						activeConnection = obj:GetPropertyChangedSignal(prop):Connect(function()
							trackedTexts[key] = tostring(obj[prop])
						end)
					end
				else
					trackedTexts[key] = nil
				end
			else
				if activeConnection then
					activeConnection:Disconnect()
					activeConnection = nil
				end
				trackedTexts[key] = nil
			end
			task.wait(CONFIG.Intervals.TextPoll)
		end
		if activeConnection then activeConnection:Disconnect() end
	end)
end

for key, path in pairs(CONFIG.TrackedTexts) do monitorText(key, path) end

local function applyAuraEffect(character)
	if not character or character:FindFirstChild("ExecutorAuraEffect") then return end
	local A = CONFIG.Aura

	new("Highlight", {
		Name = "ExecutorAuraEffect", Adornee = character, FillTransparency = 1,
		OutlineColor = State.colors.aura or A.OutlineColor, OutlineTransparency = 0,
	}, character)

	local head = character:WaitForChild("Head", 5)
	if not head then return end

	local billboard = new("BillboardGui", {
		Name = "AuraTextGui", Size = UDim2.new(0, 200, 0, 50),
		StudsOffset = Vector3.new(0, A.BobHeight, 0), AlwaysOnTop = true, Adornee = head,
	})
	local textLabel = new("TextLabel", {
		Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Font = CONFIG.Font,
		Text = State.nameText, TextSize = A.InitialTextSize, TextColor3 = State.colors.name or A.TextColor,
		TextStrokeTransparency = 0, TextStrokeColor3 = A.StrokeColor,
	}, billboard)
	local glowLabel = new("TextLabel", {
		Size = UDim2.new(1, 0, 1, 0), BackgroundTransparency = 1, Font = CONFIG.Font,
		Text = State.nameText, TextSize = A.InitialTextSize, TextColor3 = A.GlowStart,
		TextStrokeTransparency = 1, ZIndex = textLabel.ZIndex - 1,
	}, billboard)
	billboard.Parent = character
	nametag.text, nametag.glow = textLabel, glowLabel

	local startTime = tick()
	local connection
	connection = track(RunService.RenderStepped:Connect(function()
		if not character.Parent or not billboard.Parent then
			connection:Disconnect()
			return
		end
		local elapsed = tick() - startTime
		billboard.StudsOffset = Vector3.new(0, A.BobHeight + math.sin(elapsed * A.BobSpeed) * A.BobAmount, 0)
		local pulse = (math.sin(elapsed * A.PulseSpeed) + 1) / 2
		local nameColor = State.colors.name
		local glowLow = nameColor and nameColor:Lerp(Color3.new(0, 0, 0), 0.7) or A.GlowPulseLow
		glowLabel.TextColor3 = glowLow:Lerp(nameColor or A.GlowPulseHigh, pulse)
		local scale = State.nameSize / A.BaseTextSize
		billboard.Size = UDim2.new(0, 200 * scale, 0, 50 * scale)
		local size = State.nameSize + math.sin(elapsed * A.ScaleSpeed) * 0.05 * 4
		textLabel.TextSize = size
		glowLabel.TextSize = size
	end))
end

	applyAuraEffect(character)

local function makeEmitter(parent, name, overrides, extra)
	local emitter = Instance.new("ParticleEmitter")
	emitter.Name = name
	emitter.Texture = CONFIG.SSJ.Texture
	emitter.LightEmission = 0.6
	emitter.LightInfluence = 0
	for k, v in pairs(overrides) do emitter[k] = v end
	if extra then
		for k, v in pairs(extra) do emitter[k] = v end
	end
	emitter.Parent = parent
	return emitter
end

	local axis = o.axis.Unit
	local tangent = axis:Cross(Vector3.new(0, 1, 0))

end

		local highlight = character:FindFirstChild("ExecutorAuraEffect")
		if highlight then highlight.OutlineColor = colors.aura or A.OutlineColor end

		for _, partName in ipairs(S.BodyParts) do
			local part = character:FindFirstChild(partName)
			local emitter = part and part:FindFirstChild("SSJ_AuraParticles")
			if emitter then
				local base = (partName == "LeftHand" or partName == "RightHand") and EMITTER_HAND.Color or EMITTER_BODY.Color
				emitter.Color = colors.aura and recolorSequence(base, colors.aura) or base
			end
		end
	end

	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	if wants("orbs") or wants("light") then
		local soulColors
		if wants("orbs") then
			soulColors = {}
			for _, def in ipairs(SOUL_EMITTERS) do
				soulColors[def[1]] = colors.orbs and recolorSequence(def[2].Color, colors.orbs) or def[2].Color
			end
		end

		for _, att in ipairs(hrp:GetChildren()) do
			if att.Name:sub(1, 9) == "SSJ_Soul_" then
				if soulColors then
					for emitterName, sequence in pairs(soulColors) do
						local emitter = att:FindFirstChild(emitterName)
						if emitter then emitter.Color = sequence end
					end
				end
				if wants("light") then
					local light = att:FindFirstChild("SSJ_SoulLight")
					if light then light.Color = colors.light or S.LightColor end
				end
			end
		end
	end
end

local function createAura(character)
	local S = CONFIG.SSJ
	local hrp = character:WaitForChild("HumanoidRootPart", 5)
	if not hrp then return end

	if _G.SSJ_AuraConnection then
		_G.SSJ_AuraConnection:Disconnect()
		_G.SSJ_AuraConnection = nil
	end

	for _, child in ipairs(hrp:GetChildren()) do
		local n = child.Name
		if n == "SSJ_ChevronAura" or n == "SSJ_AuraAttachment" or n == "SSJ_OrbitFolder" or n:sub(1, 9) == "SSJ_Soul_" then
			child:Destroy()
		end
	end

	for _, partName in ipairs(S.BodyParts) do
		local part = character:FindFirstChild(partName)
		if part then
			local old = part:FindFirstChild("SSJ_AuraParticles")
			if old then old:Destroy() end
			local isHand = partName == "LeftHand" or partName == "RightHand"
			makeEmitter(part, "SSJ_AuraParticles", EMITTER_BODY, isHand and EMITTER_HAND or nil)
		end
	end

	new("Folder", { Name = "SSJ_OrbitFolder" }, hrp)

	local souls = {}
	for i = 1, S.SoulCount do
		local att = new("Attachment", { Name = "SSJ_Soul_" .. i }, hrp)
		for _, def in ipairs(SOUL_EMITTERS) do makeEmitter(att, def[1], def[2]) end
		new("PointLight", { Name = "SSJ_SoulLight", Color = S.LightColor, Brightness = S.LightBrightness, Range = S.LightRange }, att)
		local orbit = orbits[((i - 1) % #orbits) + 1]
		souls[i] = { att = att, orbit = orbit, phase = orbit.offset + ((i > #orbits and math.pi / 4) or 0) }
	end

	applyColors()

	local angle = 0
	local dashTable = {}
	_G.SSJ_AuraConnection = RunService.Heartbeat:Connect(function(dt)
		if not hrp.Parent then
			_G.SSJ_AuraConnection:Disconnect()
			_G.SSJ_AuraConnection = nil
			return
		end
		angle = angle + dt * S.OrbitSpeed
		if not S.PersistDash then table.clear(dashTable) end

		for i, soul in ipairs(souls) do
			local orbit = soul.orbit
			local dash = dashTable[i]
			if not dash then
				dash = { active = false, startTime = 0, duration = 0, startAngle = 0, endAngle = 0, startRadius = S.OrbitRadius, targetRadius = S.OrbitRadius }
				dashTable[i] = dash
			end

			local baseAngle = angle * orbit.speed + soul.phase

			if not dash.active and math.random() < S.DashChance then
				dash.active = true
				dash.startTime = tick()
				dash.duration = math.random(S.DashDuration[1], S.DashDuration[2]) / 100
				dash.startAngle = baseAngle
				dash.endAngle = baseAngle + (math.random() > 0.5 and 1 or -1) * (math.pi * 0.75)
				dash.startRadius = S.OrbitRadius
				dash.targetRadius = S.OrbitRadius * math.random(S.DashRadius[1], S.DashRadius[2]) / 100
			end

			local currentAngle, currentRadius = baseAngle, S.OrbitRadius
			if dash.active then
				local progress = math.clamp((tick() - dash.startTime) / dash.duration, 0, 1)
				currentAngle = dash.startAngle + (dash.endAngle - dash.startAngle) * progress
				currentRadius = dash.startRadius + (dash.targetRadius - dash.startRadius) * progress
				if progress >= 1 then dash.active = false end
			end

			local pos = orbit.tangent * (math.cos(currentAngle) * currentRadius) + orbit.bitangent * (math.sin(currentAngle) * currentRadius)
			soul.att.Position = Vector3.new(pos.X, pos.Y + 0.5, pos.Z)
		end
	end)
end

onCharacter(createAura)

local function enforceMovement()
	local character = localPlayer.Character
	if not character then return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end
	local root = character:FindFirstChild("HumanoidRootPart")

	if humanoid.WalkSpeed ~= State.walkSpeed then humanoid.WalkSpeed = State.walkSpeed end
	if humanoid.JumpPower ~= State.jumpPower then
		humanoid.UseJumpPower = true
		humanoid.JumpPower = State.jumpPower
	end
	if root and humanoid.MoveDirection.Magnitude > 0 and State.walkSpeed > CONFIG.Movement.BoostThreshold then
		local moveDir = humanoid.MoveDirection
		local vel = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(moveDir.X * State.walkSpeed, vel.Y, moveDir.Z * State.walkSpeed)
	end
end

local function getMyPlot()
	local plots = Workspace:FindFirstChild("Plots")
	if not plots then return "None" end
	local myName, myDisplay = string.lower(localPlayer.Name), string.lower(localPlayer.DisplayName)

		nameLabel = nameLabel and nameLabel:FindFirstChild("Screen")
		nameLabel = nameLabel and nameLabel:FindFirstChild("SurfaceGui")
		nameLabel = nameLabel and nameLabel:FindFirstChild("NameLabel")
		if nameLabel and nameLabel:IsA("TextLabel") then
			local text = string.lower(nameLabel.Text)
			if text == myName or text == myDisplay then return plot.Name end
		end
	end
	return "None"
end

local function isButtonActive(btn)
	if not btn or not btn.Parent then return false end
	if btn:IsA("BasePart") and (btn.Transparency >= 1 or not btn.CanCollide) then return false end
	local head = btn:FindFirstChild("Head") or btn:FindFirstChild("Part") or btn:FindFirstChildWhichIsA("BasePart")
	if head and (head.Transparency >= 1 or not head.CanCollide) then return false end
	local billboard = btn:FindFirstChildWhichIsA("BillboardGui", true)
	if billboard and not billboard.Enabled then return false end
	if btn:IsDescendantOf(ReplicatedStorage) or btn:IsDescendantOf(Lighting) then return false end
	return true
end

	if target:IsA("BasePart") then return target.Position end

		return head and head.Position or target:GetPivot().Position

end

local function moveToTarget(target)
	local character = localPlayer.Character
	if not character then return end
	local root = character:FindFirstChild("HumanoidRootPart")
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid then return end
	local targetPos = getTargetPosition(target)
	if not targetPos then return end
	cancelPath()
	if not State.usePathfind then
		root.CFrame = CFrame.new(targetPos + Vector3.new(0, 3, 0))
		return
	end
	local P = CONFIG.Pathfinding
	activePathTask = task.spawn(function()
		local path = PathfindingService:CreatePath({
			AgentRadius = P.AgentRadius, AgentHeight = P.AgentHeight, AgentCanJump = P.AgentCanJump,
			AgentCanClimb = P.AgentCanClimb, WaypointSpacing = P.WaypointSpacing,
			Costs = { Water = P.WaterCost, Danger = math.huge },
		})
		local startedAt, lastJump = tick(), 0
		local function aborted()
			return not root.Parent or humanoid.Health <= 0 or not target.Parent or not isButtonActive(target)
		end
		local function jumpIfHigher()
			local pos = getTargetPosition(target)
			if pos and pos.Y - root.Position.Y > P.JumpHeightThreshold and tick() - lastJump >= P.JumpInterval then
				lastJump = tick()
				humanoid.Jump = true
			end
		end
		local function fallbackTeleport()
			if tick() - startedAt < P.FallbackSeconds then return false end
			local pos = getTargetPosition(target)
			if pos and (root.Position - pos).Magnitude > P.ReachDistance then
				root.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
			end
			return true
		end
		while true do
			if aborted() or fallbackTeleport() then return end
			targetPos = getTargetPosition(target)
			if not targetPos then return end
			jumpIfHigher()
			if (root.Position - targetPos).Magnitude <= P.ReachDistance then
				humanoid:MoveTo(targetPos)
				task.wait(0.35)
				continue
			end
			local ok = pcall(function() path:ComputeAsync(root.Position, targetPos) end)
			if not ok or path.Status ~= Enum.PathStatus.Success then
				humanoid:MoveTo(targetPos)
				task.wait(0.45)
				continue
			end
			local waypoints = path:GetWaypoints()
			if #waypoints < 2 then
				humanoid:MoveTo(targetPos)
				task.wait(0.3)
				continue
			end
			for i = 2, #waypoints do
				if aborted() or fallbackTeleport() then return end
				local wpPos = waypoints[i].Position
				targetPos = getTargetPosition(target)
				if not targetPos then return end
				if (root.Position - targetPos).Magnitude <= P.ReachDistance then break end
				humanoid:MoveTo(wpPos)
				local startTime = tick()
				local stuckCheck, stuckTimer = root.Position, 0
				while (root.Position - wpPos).Magnitude > P.WaypointReach do
					if aborted() or fallbackTeleport() then return end
					targetPos = getTargetPosition(target)
					if targetPos and (root.Position - targetPos).Magnitude <= P.ReachDistance then break end
					if tick() - startTime > P.WaypointTimeout then break end
					jumpIfHigher()
					if (root.Position - stuckCheck).Magnitude < P.StuckDistance then
						stuckTimer += 0.08
						if stuckTimer > P.StuckTime then
							local away = root.Position - wpPos
							if away.Magnitude < 0.1 then away = Vector3.new(math.random(-1, 1), 0, math.random(-1, 1)) end
							root.CFrame = root.CFrame + (away.Unit * P.StuckPush) + Vector3.new(0, P.StuckLift, 0)
							stuckTimer = 0
							stuckCheck = root.Position
							task.wait(0.1)
						end
					else
						stuckTimer = 0
						stuckCheck = root.Position
					end
					task.wait(0.07)
				end
			end
			task.wait(0.12)
		end
	end)
end

local beams = {}
local merchantBeams = {}

local function destroyBeam(set, key)
	local data = set[key]
	if not data then return end
	data.beam:Destroy(); data.att0:Destroy(); data.att1:Destroy()
	set[key] = nil
end

local function clearBeams(set)
	for key in pairs(set) do destroyBeam(set, key) end
end

local function clearGuidelines()
	clearBeams(beams)
end

local function ensureBeam(set, key, root, targetPart, color)
	if set[key] then return end
	local G = CONFIG.Guidelines
	local att0 = Instance.new("Attachment", root)
	local att1 = Instance.new("Attachment", targetPart)
	local beam = new("Beam", {
		Attachment0 = att0, Attachment1 = att1, Color = ColorSequence.new(color or G.Color),
		Width0 = G.Width, Width1 = G.Width, Texture = G.Texture, TextureMode = Enum.TextureMode.Wrap,
		TextureSpeed = G.TextureSpeed, TextureLength = G.TextureLength, FaceCamera = true,
		Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.1), NumberSequenceKeypoint.new(0.5, 0), NumberSequenceKeypoint.new(1, 0.1) }),
	}, root)
	set[key] = { beam = beam, att0 = att0, att1 = att1 }
end

local function updateGuidelines(validButtons)
	if not State.beamsEnabled then return clearGuidelines() end
	local character = localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then return clearGuidelines() end
	local G = CONFIG.Guidelines
	local currentValid = {}
	for _, item in ipairs(validButtons) do
		if item.canAfford then
			local btn = item.instance
			currentValid[btn] = true
			local targetPart = btn:IsA("BasePart") and btn
				or (btn:IsA("Model") and (btn:FindFirstChild("Head") or btn:FindFirstChild("Part") or btn:FindFirstChildWhichIsA("BasePart") or btn.PrimaryPart))
			if targetPart then
				if (root.Position - targetPart.Position).Magnitude <= G.MaxDistance then
					ensureBeam(beams, btn, root, targetPart)
				else
					destroyBeam(beams, btn)
				end
			end
		end
	end
	for btn in pairs(beams) do
		if not currentValid[btn] then destroyBeam(beams, btn) end
	end
end

local function updateMonitor()
	local textVal = globalProgressLabel and globalProgressLabel.Text or ""
	local rawNumbersVal = globalNumbersLabel and globalNumbersLabel.Text or ""
	local numbersVal = rawNumbersVal

	if not isBlank(rawNumbersVal) then
		local current = tonumber(string.match(rawNumbersVal, "(%d+)"))
		if current then numbersVal = rebirthPair(current) end
	end

	if isBlank(numbersVal) then
		local leaderstats = localPlayer:FindFirstChild("leaderstats") or localPlayer:FindFirstChild("stats")
		local rebirthVal = leaderstats and (leaderstats:FindFirstChild("Rebirths") or leaderstats:FindFirstChild("Rebirth") or leaderstats:FindFirstChild("Rank") or leaderstats:FindFirstChild("RebirthsCount"))
		if rebirthVal then numbersVal = rebirthPair(tonumber(rebirthVal.Value) or 0) end
	end

	if isBlank(numbersVal) then
		local pGui = getPlayerGui()
		if pGui then
			for _, v in ipairs(pGui:GetDescendants()) do
				if v:IsA("TextLabel") and v.Visible then
					local nameCheck = string.lower(v.Name)
					if (string.find(nameCheck, "reb") or string.find(nameCheck, "rank")) and string.match(v.Text, "%d+") then
						local num = tonumber(string.match(v.Text, "(%d+)"))
						if num then numbersVal = rebirthPair(num); break end
					end
				end
			end
		end
	end

	if not isBlank(numbersVal) then
		if lastNumbersText == "" then
			lastNumbersText = numbersVal
		elseif numbersVal ~= lastNumbersText then
			local oldFrom = string.match(lastNumbersText, "%((%d+)%s*%-%->%s*%d+%)")
			local newFrom = string.match(numbersVal, "%((%d+)%s*%-%->%s*%d+%)")
			if oldFrom and newFrom and tonumber(newFrom) > tonumber(oldFrom) then lastNumbersText = numbersVal end
		end
	end

	local displayNumbers = (not isBlank(numbersVal)) and numbersVal or (lastNumbersText ~= "" and lastNumbersText or "N/A")
	local isCappedVisible = checkCappedState()

	RebirthLabel.Text = "Progress: " .. (textVal ~= "" and textVal or "N/A")
	CappedStatusLabel.Text = "Capped: " .. (isCappedVisible and "TRUE" or "FALSE")
	NumbersLabel.Text = "Rebirth(s): " .. displayNumbers

	local percent = parseProgress(textVal ~= "" and textVal or displayNumbers)

	if State.autoRebirthEnabled then
		local rebirthNotiText = trackedTexts["RebirthNotification"]
		local hasRebirthNoti = rebirthNotiText ~= nil and rebirthNotiText ~= "" and rebirthNotiText ~= "N/A"
		local mode = CONFIG.Rebirth.Modes[State.rebirthModeIndex].id
		local shouldRebirth
		if mode == "Any" then
			shouldRebirth = isEssenceCapped() or hasRebirthNoti
		elseif mode == "Essence" then
			shouldRebirth = isEssenceCapped()
		else
			shouldRebirth = hasRebirthNoti
		end
		if shouldRebirth then triggerRebirth() end
	end

	PercentLabel.Text = string.format("%.1f%%", percent)

	UIGradient.Rotation = 0
	if isCappedVisible then
		State.isCompleteRGB = true
	else
		State.isCompleteRGB = false
		if (State.autoRebirthEnabled and percent >= 100) or percent >= 99 then
			UIGradient.Color = ColorSequence.new(C.On)
		else
			UIGradient.Color = ColorSequence.new(Color3.fromRGB(255, 0, 0):Lerp(C.On, percent / 100))
		end
	end

	ProgressBarFill:TweenSize(UDim2.new(percent / 100, 0, 1, 0), Enum.EasingDirection.Out, Enum.EasingStyle.Quad, 0.2, true)
end

		for _, conn in ipairs({ textConn, cappedConn, numbersConn, leaderConn }) do
			if conn then conn:Disconnect() end
		end
	end
	local function hookMonitor(gui)
		if textConn then textConn:Disconnect() textConn = nil end
		if cappedConn then cappedConn:Disconnect() cappedConn = nil end
		if numbersConn then numbersConn:Disconnect() numbersConn = nil end
		pcall(function()
			globalProgressLabel = gui:WaitForChild("System", 2):WaitForChild("RebirthProgress", 2):WaitForChild("Progress", 2):WaitForChild("TextLabel", 2)
		end)
		pcall(function()
			local rebirthFrame = gui:FindFirstChild("Main") and gui.Main:FindFirstChild("Rebirth")
			if rebirthFrame then globalCappedObj = rebirthFrame:FindFirstChild("Capped", true) end
			if not globalCappedObj then
				local starterRebirth = StarterGui:FindFirstChild("Main") and StarterGui.Main:FindFirstChild("Rebirth")
				if starterRebirth then globalCappedObj = starterRebirth:FindFirstChild("Capped", true) end
			end
		end)
		pcall(function()
			local holder = gui:WaitForChild("System", 2):WaitForChild("PlayerList", 2):WaitForChild("Holder", 2)
			local myHolder = holder:FindFirstChild(localPlayer.Name) or holder:FindFirstChildWhichIsA("Frame")
			if myHolder then globalNumbersLabel = myHolder:WaitForChild("RebsText", 2) end
		end)
		pcall(function()
			if not globalNumbersLabel then
				local rebirthFrame = gui:FindFirstChild("Main") and gui.Main:FindFirstChild("Rebirth")
				if rebirthFrame then globalNumbersLabel = rebirthFrame:FindFirstChild("Numbers", true) end
			end
		end)
		if globalProgressLabel then textConn = globalProgressLabel:GetPropertyChangedSignal("Text"):Connect(updateMonitor) end
		if globalCappedObj then cappedConn = globalCappedObj:GetPropertyChangedSignal("Visible"):Connect(updateMonitor) end
		if globalNumbersLabel then numbersConn = globalNumbersLabel:GetPropertyChangedSignal("Text"):Connect(updateMonitor) end
		local leaderstats = localPlayer:FindFirstChild("leaderstats") or localPlayer:FindFirstChild("stats")
		local cashVal = leaderstats and (leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Money"))
		if cashVal then
			if leaderConn then leaderConn:Disconnect() end
			leaderConn = cashVal:GetPropertyChangedSignal("Value"):Connect(updateMonitor)
		end
		updateMonitor()
	end
	hookMonitor(localPlayer:WaitForChild("PlayerGui"))
	track(localPlayer.CharacterAdded:Connect(function()
		task.wait(1)
		if running then hookMonitor(localPlayer:WaitForChild("PlayerGui")) end
	end))
	while running do
		task.wait(CONFIG.Intervals.MonitorPoll)
		if running then updateMonitor() end
	end
end)

-- autobuy settings thingy (fixed this up)
local existingUIButtons = {}
local lastAutoBuyTime = 0

local function clearShopList()
	for _, uiBtn in pairs(existingUIButtons) do uiBtn:Destroy() end
	table.clear(existingUIButtons)
	clearGuidelines()
end

local function updateShop()
	local cashNum, cashStr = getPlayerCash()
	local currentPlot = getMyPlot()

	StatsLabel.Text = string.format("Stats:\nCash: %s\nPlot: %s\nEssence Capped: %s", cashStr or tostring(cashNum), currentPlot, isEssenceCapped() and "True" or "False")

	local closestTycoon = getClosestTycoon()
	if not closestTycoon or not closestTycoon:FindFirstChild("Buttons") then
		clearShopList()
		return
	end

	local affordable, unaffordable, allItems = {}, {}, {}
	for _, btn in ipairs(closestTycoon.Buttons:GetChildren()) do
		local priceNum, priceStr, displayName = getButtonDetails(btn)
		if priceNum then
			local item = {
				instance = btn, name = displayName or btn.Name, price = priceNum,
				priceStr = priceStr or tostring(priceNum), canAfford = cashNum >= priceNum,
			}
			table.insert(allItems, item)
			table.insert(item.canAfford and affordable or unaffordable, item)
		end
	end

	local function byPrice(a, b) return a.price < b.price end
	table.sort(affordable, byPrice)
	table.sort(unaffordable, byPrice)

	if State.autoBuyEnabled and currentPlot ~= "None" and localPlayer.Character then
		if #affordable > 0 and tick() - lastAutoBuyTime >= CONFIG.Intervals.AutoBuyCooldown then
			lastAutoBuyTime = tick()
			moveToTarget(affordable[1].instance)
		end
	end

	local currentValid = {}
	local order = 1
	local function processList(list)
		for _, item in ipairs(list) do
			currentValid[item.instance] = true
			local itemBtn = existingUIButtons[item.instance]
			if not itemBtn then
				itemBtn = new("TextButton", { Name = item.name, Size = UDim2.new(1, 0, 0, 32), Font = CONFIG.Font, TextWrapped = true }, MainFrame)
				itemBtn.Visible = false  -- tracked but never rendered
				addCorner(itemBtn, 6)
				addStroke(itemBtn, C.Stroke, 1)
				existingUIButtons[item.instance] = itemBtn
			end
			itemBtn.LayoutOrder = order
			itemBtn.Text = item.name .. " (" .. item.priceStr .. ")"
			itemBtn.BackgroundColor3 = item.canAfford and C.On or C.Danger
			itemBtn.TextColor3 = C.Text
			if itemBtn._conn then itemBtn._conn:Disconnect() end
			itemBtn._conn = itemBtn.MouseButton1Click:Connect(function()
				if item.canAfford then moveToTarget(item.instance) end
			end)
			order += 1
		end
	end
	processList(affordable)
	processList(unaffordable)

	for btn, uiBtn in pairs(existingUIButtons) do
		if not currentValid[btn] then
			if uiBtn._conn then uiBtn._conn:Disconnect() end
			uiBtn:Destroy()
			existingUIButtons[btn] = nil
		end
	end
	updateGuidelines(allItems)
end

--  scheduler
local gradientOffset = 0

local scheduled = {
	{ interval = CONFIG.Intervals.MovementCheck, last = 0, fn = enforceMovement },
	{ interval = CONFIG.Intervals.GradientSpin, last = 0, fn = function()
		if not State.isCompleteRGB then return end
		gradientOffset = (gradientOffset + 0.02) % 1
		local keys = {}
		for i = 0, 5 do
			local t = i / 5
			keys[#keys + 1] = ColorSequenceKeypoint.new(t, Color3.fromHSV((t + gradientOffset) % 1, 1, 1))
		end
		UIGradient.Color = ColorSequence.new(keys)
	end },
	{ interval = CONFIG.Intervals.RgbUpdate, last = 0, fn = function(now)
		local period = CONFIG.Rgb.CycleSeconds / State.rgbSpeed
		local rgb = Color3.fromHSV((now % period) / period, 1, 1)
		Title.Text = string.format('%s  <font color="rgb(%d,%d,%d)">%s</font>',
			CONFIG.Title.Prefix, math.floor(rgb.R * 255), math.floor(rgb.G * 255), math.floor(rgb.B * 255), CONFIG.Title.Tag)
		for key, on in pairs(State.rgb) do
			if on then
				State.colors[key] = rgb
				applyColors(key)
			end
		end
	end },
	{ interval = CONFIG.Intervals.MerchantUpdate, last = 0, fn = highlightMerchants },
	{ interval = CONFIG.Intervals.ShopUpdate, last = 0, fn = updateShop },
	{ interval = CONFIG.Intervals.MerchantLocator, last = 0, fn = updateMerchantLocator },
}

track(RunService.Heartbeat:Connect(function()
	local now = tick()
	for _, job in ipairs(scheduled) do
		if now - job.last >= job.interval then
			job.last = now
			job.fn(now)
		end
	end
end))

--  fix on re-exec
local function clearAuras(character)
	for _, name in ipairs({ "ExecutorAuraEffect", "AuraTextGui" }) do
		local obj = character:FindFirstChild(name)
		if obj then obj:Destroy() end
	end
	for _, partName in ipairs(CONFIG.SSJ.BodyParts) do
		local part = character:FindFirstChild(partName)
		local emitter = part and part:FindFirstChild("SSJ_AuraParticles")
		if emitter then emitter:Destroy() end
	end
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if hrp then
		for _, child in ipairs(hrp:GetChildren()) do
			local n = child.Name
			if n == "SSJ_OrbitFolder" or n:sub(1, 9) == "SSJ_Soul_" then child:Destroy() end
		end
	end
end

env.TycoonManagerCleanup = function()
	running = false
	for _, connection in ipairs(connections) do connection:Disconnect() end
	table.clear(connections)
	if speedConnection then speedConnection:Disconnect() speedConnection = nil end
	if _G.SSJ_AuraConnection then
		_G.SSJ_AuraConnection:Disconnect()
		_G.SSJ_AuraConnection = nil
	end
	for _, hook in ipairs(cleanupHooks) do pcall(hook) end
	table.clear(cleanupHooks)
	pcall(cancelPath)
	pcall(clearGuidelines)
	pcall(clearBeams, merchantBeams)
	local character = localPlayer.Character
	if character then
		pcall(clearAuras, character)
		local humanoid = character:FindFirstChildOfClass("Humanoid")
		if humanoid and originalMovement then
			humanoid.WalkSpeed = originalMovement.WalkSpeed
			humanoid.UseJumpPower = originalMovement.UseJumpPower
			humanoid.JumpPower = originalMovement.JumpPower
		end
	end
	ScreenGui:Destroy()
	Sound:Stop()
	Sound:Destroy()
	env.TycoonManagerCleanup = nil
end
